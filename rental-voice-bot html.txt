<script>
  const chat = document.getElementById('chat');
  const status = document.getElementById('status');
  const micBtn = document.getElementById('micBtn');

  // Check Web Speech API Support
  const SpeechRecognition = window.SpeechRecognition || window.webkitSpeechRecognition;
  let recognition;

  if (SpeechRecognition) {
    recognition = new SpeechRecognition();
    recognition.continuous = false;
    recognition.lang = 'en-US';

    recognition.onstart = () => {
      micBtn.classList.add('listening');
      status.innerText = "Listening... Speak now";
    };

    recognition.onend = () => {
      micBtn.classList.remove('listening');
      status.innerText = "Click microphone to start talking";
    };

    recognition.onresult = (event) => {
      const transcript = event.results[0][0].transcript;
      handleUserQuery(transcript);
    };

    recognition.onerror = (e) => {
      micBtn.classList.remove('listening');
      status.innerText = "Error recognizing voice. Try again.";
    };
  } else {
    status.innerText = "Speech Recognition not supported in this browser.";
    micBtn.style.opacity = 0.5;
  }

  micBtn.addEventListener('click', () => {
    if (!SpeechRecognition) return;
    if (micBtn.classList.contains('listening')) {
      recognition.stop();
    } else {
      window.speechSynthesis.cancel(); // Stop any ongoing bot speech
      recognition.start();
    }
  });

  function simulatePrompt(text) {
    window.speechSynthesis.cancel();
    handleUserQuery(text);
  }

  function handleUserQuery(query) {
    addMessage(query, 'user');
    const response = getBotResponse(query);
    setTimeout(() => {
      addMessage(response, 'bot');
      speakText(response);
    }, 400);
  }

  function addMessage(text, sender) {
    const msg = document.createElement('div');
    msg.className = `msg ${sender}`;
    msg.innerText = text;
    chat.appendChild(msg);
    chat.scrollTop = chat.scrollHeight;
  }

  function speakText(text) {
    if ('speechSynthesis' in window) {
      const utterance = new SpeechSynthesisUtterance(text);
      utterance.rate = 1;
      utterance.pitch = 1;
      window.speechSynthesis.speak(utterance);
    }
  }

  // Knowledge Base Engine for Rental Support
  function getBotResponse(input) {
    const text = input.toLowerCase();

    if (text.includes('rate') || text.includes('price') || text.includes('cost') || text.includes('how much')) {
      return "Economy cars start at $35 per day, SUVs at $65 per day, and luxury vehicles start at $100 per day. All rentals include unlimited mileage.";
    }
    if (text.includes('extend') || text.includes('longer') || text.includes('keep')) {
      return "You can extend your booking directly through our mobile app or by providing me your booking ID right now. Extensions depend on vehicle availability.";
    }
    if (text.includes('deposit') || text.includes('security') || text.includes('hold')) {
      return "We require a refundable security deposit of $200 on a credit or debit card, which is released within 3 to 5 business days after returning the vehicle safely.";
    }
    if (text.includes('accident') || text.includes('breakdown') || text.includes('emergency') || text.includes('flat tire')) {
      return "If you are in immediate danger, please call emergency services first. For 24/7 DriveEasy Roadside Assistance, stay on the line or call our helpline at 1-800-DRIVE-SAFE.";
    }
    if (text.includes('cancel') || text.includes('refund')) {
      return "Cancellations made 24 hours prior to pickup receive a full 100% refund. Cancellations within 24 hours incur a 1-day rental fee.";
    }
    if (text.includes('age') || text.includes('old') || text.includes('license')) {
      return "Drivers must be at least 21 years old with a valid driver's license held for at least 1 year. Drivers under 25 may incur a young driver fee.";
    }

    return "I'm sorry, I didn't quite catch that. You can ask me about rental rates, booking extensions, security deposits, or cancellation policies!";
  }
</script>
